461[[nodiscard]] inline std::uint64_t multiply_high(std::uint64_t a, std::uint64_t b) noexcept {
462 const std::uint64_t a_low = a & 0xFFFFFFFFULL;
463 const std::uint64_t a_high = a >> 32U;
464 const std::uint64_t b_low = b & 0xFFFFFFFFULL;
465 const std::uint64_t b_high = b >> 32U;
466 const std::uint64_t low_product = a_low * b_low;
467 const std::uint64_t middle = a_high * b_low + (low_product >> 32U);
468 const std::uint64_t upper = a_low * b_high + (middle & 0xFFFFFFFFULL);
469 return a_high * b_high + (middle >> 32U) + (upper >> 32U);
470}

476[[nodiscard]] inline PermitId fnv1a_128(std::span<const std::uint8_t> bytes) noexcept {
477 std::uint64_t high = kFnvOffsetHigh;
478 std::uint64_t low = kFnvOffsetLow;
479 for (const std::uint8_t byte : bytes) {
480 low ^= static_cast<std::uint64_t>(byte);
481 const std::uint64_t carry = multiply_high(low, kFnvPrimeLow);
482 const std::uint64_t next_high = carry + (high * kFnvPrimeLow) + (low * kFnvPrimeHigh);
483 high = next_high;
484 low *= kFnvPrimeLow;
485 }
486 PermitId digest{};
487 for (std::size_t index = 0; index < 8; ++index) {
488 const std::size_t shift = index * 8U;
489 digest[index] = static_cast<std::uint8_t>((low >> shift) & 0xFFU);
490 digest[index + 8U] = static_cast<std::uint8_t>((high >> shift) & 0xFFU);
491 }
492 return digest;
493}

1224 Action action) noexcept {
